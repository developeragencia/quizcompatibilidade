        function calculateAgeAndSign() {
            const birthdateInput = document.getElementById('registerBirthdate');
            const birthInfo = document.getElementById('birthInfo');
            
            if (!birthdateInput.value) {
                birthInfo.classList.add('hidden');
                return;
            }

            const birthdate = new Date(birthdateInput.value);
            const today = new Date();
            
            // Calcular idade
            let age = today.getFullYear() - birthdate.getFullYear();
            const monthDiff = today.getMonth() - birthdate.getMonth();
            if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birthdate.getDate())) {
                age--;
            }
            
            // Determinar signo
            const month = birthdate.getMonth() + 1;
            const day = birthdate.getDate();
            const sign = getZodiacSign(month, day);

            document.getElementById('calculatedSign').textContent = sign;
            document.getElementById('dailyQuote').textContent = randomQuote;
            
            birthInfo.classList.remove('hidden');
        }

        // Função para determinar o signo
        function getZodiacSign(month, day) {
            const signs = [
                { sign: 'Capricórnio ♑', start: [12, 22], end: [1, 19] },
                { sign: 'Aquário ♒', start: [1, 20], end: [2, 18] },
                { sign: 'Peixes ♓', start: [2, 19], end: [3, 20] },
                { sign: 'Áries ♈', start: [3, 21], end: [4, 19] },
                { sign: 'Touro ♉', start: [4, 20], end: [5, 20] },
                { sign: 'Gêmeos ♊', start: [5, 21], end: [6, 20] },
                { sign: 'Câncer ♋', start: [6, 21], end: [7, 22] },
                { sign: 'Leão ♌', start: [7, 23], end: [8, 22] },
                { sign: 'Virgem ♍', start: [8, 23], end: [9, 22] },
                { sign: 'Libra ♎', start: [9, 23], end: [10, 22] },
                { sign: 'Escorpião ♏', start: [10, 23], end: [11, 21] },
                { sign: 'Sagitário ♐', start: [11, 22], end: [12, 21] }
            ];

            for (let i = 0; i < signs.length; i++) {
                const s = signs[i];
                const [startMonth, startDay] = s.start;
                const [endMonth, endDay] = s.end;
                
                if (startMonth === endMonth) {
                    if (month === startMonth && day >= startDay && day <= endDay) {
                        return s.sign;
                    }
                } else {
                    if ((month === startMonth && day >= startDay) || 
                        (month === endMonth && day <= endDay)) {
                        return s.sign;
                    }
                }
            }
            return 'Signo não identificado';
        }

        // Funções de autenticação
        function doLogin() {
            const instagram = document.getElementById('loginInstagram').value;
            const password = document.getElementById('loginPassword').value;

            if (!instagram || !password) {
                showNotification('Preencha todos os campos', 'error');
                return;
            }

            const user = database.users.find(u => u.instagram === instagram && u.password === password);
            
            if (user) {
                currentUser = user;
                showNotification('Login realizado com sucesso!', 'success');
                
                if (user.isAdmin) {
                    showAdminDashboard();
                } else if (user.answers) {
                    showUserDashboard();
                } else {
                    startQuestionnaire();
                }
            } else {
                showNotification('Instagram ou senha incorretos', 'error');
            }
        }

        function doRegister() {
            const name = document.getElementById('registerName').value;
            const instagram = document.getElementById('registerInstagram').value;
            const password = document.getElementById('registerPassword').value;
            const birthdateInput = document.getElementById('registerBirthdate').value;
            const accept = document.getElementById('registerAccept').checked;

            if (!name || !instagram || !password || !birthdateInput) {
                showNotification('Preencha todos os campos', 'error');
                return;
            }

            if (!accept) {
                showNotification('Você precisa aceitar os termos para continuar', 'error');
                return;
            }

            if (database.users.find(u => u.instagram === instagram)) {
                showNotification('Este Instagram já está cadastrado', 'error');
                return;
            }

            // Calcular idade a partir da data de nascimento
            const birthdate = new Date(birthdateInput);
            const today = new Date();
            let age = today.getFullYear() - birthdate.getFullYear();
            const monthDiff = today.getMonth() - birthdate.getMonth();
            if (monthDiff < 0 || (monthDiff === 0 && today.getDate() < birthdate.getDate())) {
                age--;
            }

            // Calcular signo
            const month = birthdate.getMonth() + 1;
            const day = birthdate.getDate();
            const sign = getZodiacSign(month, day);

            const newUser = {
                id: database.nextId++,
                name,
                instagram,
                password,
                age: age,
                birthdate: birthdateInput,
                sign: sign,
                isAdmin: false,
                answers: null,
                score: null
            };

            database.users.push(newUser);
            currentUser = newUser;
            
            showNotification('Cadastro realizado com sucesso!', 'success');
            startQuestionnaire();
        }

        // Funções do questionário
        function startQuestionnaire() {
            hideAllPages();
            document.getElementById('questionnairePage').classList.remove('hidden');
            currentQuestionIndex = 0;
            userAnswers = {};
            displayQuestion();
        }

        function displayQuestion() {
            const question = questions[currentQuestionIndex];
            const container = document.getElementById('questionContainer');
            
            container.innerHTML = `
                <div class="question">
                    <h3>${question.text}</h3>
                    <div class="radio-group">
                        ${question.options.map((option, index) => `
                            <div class="radio-option">
                                <input type="radio" name="q${question.id}" value="${option.value}" id="opt${index}">
                                <label for="opt${index}">${option.text}</label>
                            </div>
                        `).join('')}
                    </div>
                </div>
            `;

            document.getElementById('currentQuestion').textContent = currentQuestionIndex + 1;
            document.getElementById('progressFill').style.width = `${((currentQuestionIndex + 1) / questions.length) * 100}%`;

            if (currentQuestionIndex === questions.length - 1) {
                document.getElementById('nextBtn').classList.add('hidden');
                document.getElementById('finishBtn').classList.remove('hidden');
            }
        }

        function nextQuestion() {
            const selectedOption = document.querySelector(`input[name="q${questions[currentQuestionIndex].id}"]:checked`);
            
            if (!selectedOption) {
                showNotification('Selecione uma resposta', 'error');
                return;
            }

            userAnswers[questions[currentQuestionIndex].id] = parseInt(selectedOption.value);
            currentQuestionIndex++;
            
            if (currentQuestionIndex < questions.length) {
                displayQuestion();
            }
        }

        function finishQuestionnaire() {
            const selectedOption = document.querySelector(`input[name="q${questions[currentQuestionIndex].id}"]:checked`);
            
            if (!selectedOption) {
                showNotification('Selecione uma resposta', 'error');
                return;
            }

            userAnswers[questions[currentQuestionIndex].id] = parseInt(selectedOption.value);
            
            // Calcular pontuação
            const score = calculateCompatibility();
            currentUser.answers = userAnswers;
            currentUser.score = score;
            
            showUserDashboard();
        }

        function calculateCompatibility() {
            let totalScore = 0;
            let maxPossibleScore = 0;

            questions.forEach(question => {
                const userAnswer = userAnswers[question.id];
                const maxScore = Math.max(...question.options.map(opt => opt.value));
                
                totalScore += userAnswer;
                maxPossibleScore += maxScore;
            });

            // Converter para porcentagem, considerando que algumas respostas podem ser negativas
            const percentage = Math.max(0, Math.round((totalScore / maxPossibleScore) * 100));
            return Math.min(100, percentage);
        }
